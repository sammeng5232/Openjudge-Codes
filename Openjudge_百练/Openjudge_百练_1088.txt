百练-1088:滑雪

总时间限制: 1000ms 内存限制: 65536kB
描述
Michael喜欢滑雪百这并不奇怪， 因为滑雪的确很刺激。可是为了获得速度，滑的区域必须向下倾斜，而且当你滑到坡底，你不得不再次走上坡或者等待升降机来载你。Michael想知道载一个区域中最长的滑坡。区域由一个二维数组给出。数组的每个数字代表点的高度。下面是一个例子
 1  2  3  4 5
16 17 18 19 6
15 24 25 20 7
14 23 22 21 8
13 12 11 10 9

一个人可以从某个点滑向上下左右相邻四个点之一，当且仅当高度减小。在上面的例子中，一条可滑行的滑坡为24-17-16-1。当然25-24-23-...-3-2-1更长。事实上，这是最长的一条。
输入
输入的第一行表示区域的行数R和列数C(1 <= R,C <= 100)。下面是R行，每行有C个整数，代表高度h，0<=h<=10000。
输出
输出最长区域的长度。
样例输入
5 5
1 2 3 4 5
16 17 18 19 6
15 24 25 20 7
14 23 22 21 8
13 12 11 10 9
样例输出
25

#include <bits/stdc++.h>
using namespace std;

int r, c;

struct pt {
  int x;
  int y;
  int h;
} p[10000];

bool cmp(pt p1, pt p2) { return p1.h < p2.h; }

bool legal(int x, int y) { return (x >= 0 && y >= 0 && x < r && y < c); }

int main() {
  int x1, y1;
  cin >> r >> c;
  int a[100][100], dp[100][100];
  int ml = 1, k = 0;
  int d[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
  for (int i = 0; i < r; i++) {
    for (int j = 0; j < c; j++) {
      cin >> a[i][j];
      p[k].x = i;
      p[k].y = j;
      p[k].h = a[i][j];
      dp[i][j] = 1;
      k++;
    }
  }
  sort(p, p + k, cmp);
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < 4; j++) {
      if (legal(p[i].x + d[j][0], p[i].y + d[j][1]) &&
          p[i].h > a[p[i].x + d[j][0]][p[i].y + d[j][1]]) {
        dp[p[i].x][p[i].y] =
            max(dp[p[i].x][p[i].y], dp[p[i].x + d[j][0]][p[i].y + d[j][1]] + 1);
        ml = max(ml, dp[p[i].x][p[i].y]);
      }
    }
  }
  cout << ml << "\n";
  return 0;
}
